<script>
(function () {
  function moveSwatches() {
    var swatchWrap = document.getElementById('beanbag-swatch-wrapper');
    var optionsTable = document.getElementById('options_table');

    if (!swatchWrap || !optionsTable) return;

    if (!swatchWrap.dataset.moved) {
      optionsTable.parentNode.insertBefore(
        swatchWrap,
        optionsTable.nextSibling
      );
      swatchWrap.dataset.moved = '1';
    }
  }

  var tries = 0;
  var timer = setInterval(function () {
    tries++;
    moveSwatches();
    if (tries > 20) clearInterval(timer);
  }, 100);
})();
</script>

<script>
(function () {

  // Covers not available when Size = King (ALREADY NORMALIZED)
  var kingExcludedCovers = [
    'faux fur / pink',
    'faux fur / navy'
  ];

  function normalize(str) {
    return str
      .toLowerCase()
      .replace(/\u00a0/g, ' ')
      .replace(/\s+/g, ' ')
      .trim();
  }

  function getSizeSelect() {
    return Array.from(
      document.querySelectorAll('#options_table select')
    ).find(sel =>
      Array.from(sel.options).some(opt =>
        normalize(opt.text).includes('king')
      )
    );
  }

  function updateSwatchesForSize() {
    var sizeSelect = getSizeSelect();
    if (!sizeSelect) return;

    var selectedSize = normalize(
      sizeSelect.options[sizeSelect.selectedIndex].text
    );

    document.querySelectorAll('.beanbag-swatch').forEach(function (swatch) {
      var cover = normalize(swatch.dataset.option);

      if (selectedSize.includes('king') && kingExcludedCovers.includes(cover)) {
        swatch.style.display = 'none';

        if (swatch.classList.contains('active')) {
          swatch.classList.remove('active');
          var label = document.getElementById('beanbag-selected-cover-name');
          if (label) label.textContent = '';
        }
      } else {
        swatch.style.display = '';
      }
    });
  }

  document.addEventListener('DOMContentLoaded', updateSwatchesForSize);

  document.addEventListener('change', function (e) {
    if (e.target.closest('#options_table')) {
      updateSwatchesForSize();
    }
  });

})();
</script>
